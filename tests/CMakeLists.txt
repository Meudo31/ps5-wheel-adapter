cmake_minimum_required(VERSION 3.13)
project(brake_tests C)
enable_testing()

foreach(test_name brake_test adapter_test g27_test g27_leds_test ffb_test pedal_test auth_detect_test status_leds_test)
    add_executable(${test_name} ${test_name}.c ../src/brake.c ../src/g27.c ../src/g27_leds.c ../src/ffb.c ../src/profile_store.c ../src/pedal.c ../src/auth_detect.c ../src/status_leds.c)
    target_include_directories(${test_name} PRIVATE ../src stubs)
    target_compile_features(${test_name} PRIVATE c_std_11)
    target_compile_definitions(${test_name} PRIVATE HOST_TEST=1)
    if(MSVC)
        target_compile_options(${test_name} PRIVATE /UNDEBUG)
    else()
        target_compile_options(${test_name} PRIVATE -UNDEBUG -Wall -Wextra)
    endif()
    add_test(NAME ${test_name} COMMAND ${test_name})
endforeach()

add_executable(brake_inverted_test brake_test.c ../src/brake.c)
target_include_directories(brake_inverted_test PRIVATE ../src)
target_compile_features(brake_inverted_test PRIVATE c_std_11)
target_compile_definitions(brake_inverted_test PRIVATE HOST_TEST=1)
target_compile_definitions(brake_inverted_test PRIVATE BRAKE_RELEASED_RAW=20000 BRAKE_PRESSED_RAW=-20000)
if(MSVC)
    target_compile_options(brake_inverted_test PRIVATE /UNDEBUG)
else()
    target_compile_options(brake_inverted_test PRIVATE -UNDEBUG -Wall -Wextra)
endif()
add_test(NAME brake_inverted_test COMMAND brake_inverted_test)
